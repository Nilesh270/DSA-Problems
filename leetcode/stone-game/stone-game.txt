class Solution {
    public int solve(int[] piles, int start, int end, int[][]dp){
        if(start==end){
            return piles[start];
        }

        if(dp[start][end]!=-1){
            return dp[start][end];
        }

        int left = piles[start] - solve(piles,start+1,end,dp);
        int right = piles[end] - solve(piles,start,end-1,dp);

        return dp[start][end] =Math.max(left,right);
    }
    public boolean stoneGame(int[] piles) {
        int n = piles.length;
        int[][] dp = new int[n][n];

        for(int[] temp : dp){
            Arrays.fill(temp,-1);
        }
        return solve(piles,0,piles.length-1,dp)>0;
    }
}
